{%- comment -%} Thumbnail URL for a post card: front matter `image`, else the first in the post, else a YouTube thumbnail for an embedded video, else nothing (the card shows a text tile). Usage: {% capture thumb %}{% include post-thumb.html post=post %}{% endcapture %} {%- endcomment -%} {%- assign _p = include.post -%} {%- assign _src = _p.image.path | default: _p.image -%} {%- unless _src -%} {%- assign _parts = _p.content | split: ' 1 -%} {%- assign _src = _parts[1] | split: 'src="' | slice: 1, 1 | first | split: '"' | first -%} {%- endif -%} {%- endunless -%} {%- unless _src -%} {%- assign _parts = _p.content | split: 'youtube.com/embed/' -%} {%- if _parts.size > 1 -%} {%- assign _yt = _parts[1] | split: '"' | first | split: '?' | first -%} {%- capture _src -%}https://i.ytimg.com/vi/{{ _yt }}/hqdefault.jpg{%- endcapture -%} {%- endif -%} {%- endunless -%} {%- if _src -%} {%- if _src contains '://' -%}{{ _src }}{%- else -%}{{ _src | relative_url }}{%- endif -%} {%- endif -%}
